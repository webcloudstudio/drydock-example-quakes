#!/bin/sh
# run.sh — Quakes grader. Do not filter, skip, or reinterpret.
#
# Runs ./quakes over the frozen USGS snapshot and compares every report's exit code and standard
# output with the known answer below. Standard error is never compared. No network is used.
#
#   sh sources/run.sh                from the application root (how `drydock uat` runs it)
#   sh uat/Quakes/bin/run.sh APP     from anywhere, grading the application root APP
#
# Exit 0 only when every check passes.
set -u

HERE=$(cd "$(dirname "$0")" && pwd)
APP=$(cd "${1:-.}" && pwd)
if [ -f "$HERE/quakes_2025-01-01_07.geojson" ]; then DATA=$HERE; else DATA=$(cd "$HERE/../sources" && pwd); fi
WORK=$(mktemp -d)
trap 'rm -rf "$WORK"' EXIT INT TERM
BIN="$APP/quakes"
QUAKES="$DATA/quakes_2025-01-01_07.geojson"
printf 'this is not json\n' > "$WORK/garbage.json"
printf '{"type":"Feature","properties":{},"geometry":null}\n' > "$WORK/feature.json"

if [ ! -x "$BIN" ]; then
    echo "error: no executable ./quakes at the application root $APP." >&2
    echo "The deliverable is an executable named quakes at the application root." >&2
    exit 1
fi

PASSED=0
FAILED=0

# check <id> <expected exit> <quakes args...>  — expected standard output on stdin.
check() {
    id=$1
    want_rc=$2
    shift 2
    cat > "$WORK/want"
    (cd "$APP" && "$BIN" "$@") > "$WORK/got" 2> "$WORK/err"
    rc=$?
    if [ "$rc" -eq "$want_rc" ] && cmp -s "$WORK/want" "$WORK/got"; then
        PASSED=$((PASSED + 1))
        echo "PASS $id: quakes $*"
    else
        FAILED=$((FAILED + 1))
        echo "FAIL $id: quakes $*"
        echo "  exit: expected $want_rc, got $rc"
        diff "$WORK/want" "$WORK/got" | sed 's/^/  /'
        sed 's/^/  stderr: /' "$WORK/err"
    fi
}
